"""
Точка сборки слоёв (composition root).

Единственное место, которое знает, какая реализация репозитория используется.
Для замены хранилища достаточно переключить REPO_BACKEND:
service.py и router.py при этом не меняются.

    REPO_BACKEND=postgres  (по умолчанию) - PostgreSQL через SQLAlchemy
    REPO_BACKEND=memory                    - хранение в памяти

import os

from fastapi import Depends
from sqlalchemy.orm import Session

from app.database import get_db
from app.memory_store import memory_store
from app.modules.delivery.repository import (
    InMemoryDeliveryRepository,
    SqlDeliveryRepository,
)

    SqlOrderRepository,
)

from app.modules.users.repository import InMemoryUserRepository, SqlUserRepository
from app.modules.users.service import UserService


def use_memory_backend() -> bool:
    return os.getenv("REPO_BACKEND", "postgres").lower() == "memory"


def get_user_service(db: Session = Depends(get_db)) -> UserService:
    if use_memory_backend():
        return UserService(InMemoryUserRepository(memory_store))
    return UserService(SqlUserRepository(db))


def get_menu_service(db: Session = Depends(get_db)) -> MenuService:
    if use_memory_backend():
        return MenuService(InMemoryMenuRepository(memory_store))
    return MenuService(SqlMenuRepository(db))


def get_order_service(db: Session = Depends(get_db)) -> OrderService:
    if use_memory_backend():
        return OrderService(InMemoryOrderRepository(memory_store))
    return OrderService(SqlOrderRepository(db))


def get_delivery_service(db: Session = Depends(get_db)) -> DeliveryService:
    if use_memory_backend():
        return DeliveryService(InMemoryDeliveryRepository(memory_store))
    return DeliveryService(SqlDeliveryRepository(db))

    """Генератор сессии базы данных для использования в Depends."""

        db.close()

    if isinstance(error, NotFoundError):
        status = 404
    elif isinstance(error, ForbiddenError):
        status = 403
    else:
        status = 400
    return HTTPException(status_code=status, detail=str(error))

app/memory_store.py:
"""
"""
Хранилище в памяти для InMemory-репозиториев (тесты и запуск без PostgreSQL).
Общее для всех модулей, чтобы orders видел пользователей и блюда.
"""
from app.entities import Delivery, MenuItem, Order, User


class MemoryStore:
    def __init__(self):
        self.users: dict[int, User] = {}
        self.menu_items: dict[int, MenuItem] = {}
        self.orders: dict[int, Order] = {}
        self.deliveries: dict[int, Delivery] = {}

    @staticmethod
    def next_id(table: dict) -> int:
        return max(table, default=0) + 1

    def clear(self) -> None:
        self.users.clear()
        self.menu_items.clear()
        self.orders.clear()
        self.deliveries.clear()


# Общий экземпляр для приложения (REPO_BACKEND=memory)
memory_store = MemoryStore()

from sqlalchemy import Column, Integer, String, Float, Boolean, ForeignKey
from sqlalchemy.orm import relationship

from app.database import Base


class User(Base):
    __tablename__ = "users"

    id = Column(Integer, primary_key=True, index=True)
    email = Column(String, unique=True, nullable=False)
    phone = Column(String, nullable=True)
    balance = Column(Float, default=0.0)
    is_blocked = Column(Boolean, default=False)


class MenuItem(Base):
    __tablename__ = "menu_items"

    id = Column(Integer, primary_key=True, index=True)
    name = Column(String, nullable=False)
    price = Column(Float, nullable=False)
    is_available = Column(Boolean, default=True)


class Order(Base):
    __tablename__ = "orders"

    id = Column(Integer, primary_key=True, index=True)
    user_id = Column(Integer, ForeignKey("users.id"))
    item_id = Column(Integer, ForeignKey("menu_items.id"))
    quantity = Column(Integer, nullable=False)
    status = Column(String, default="processing")

    user = relationship("User")
    item = relationship("MenuItem")


class Delivery(Base):

    id = Column(Integer, primary_key=True, index=True)
    order_id = Column(Integer, ForeignKey("orders.id"))
    courier_id = Column(Integer, ForeignKey("users.id"), nullable=True)
    status = Column(String, default="pending")

    order = relationship("Order")
    courier = relationship("User")

"""
"""
Инфраструктурный слой модуля delivery (Круг 3).
Абстрактный интерфейс и две реализации: SQL и in-memory.
"""
from abc import ABC, abstractmethod

from sqlalchemy.orm import Session

from app.entities import Delivery, Order, User
from app.mappers import to_delivery, to_order, to_user
from app.memory_store import MemoryStore
from app.models import Delivery as DeliveryModel
from app.models import Order as OrderModel
from app.models import User as UserModel


class DeliveryRepository(ABC):
    """Интерфейс доступа к данным доставки."""

    @abstractmethod
    def get_order_by_id(self, order_id: int) -> Order | None: ...

    @abstractmethod
    def get_user_by_id(self, user_id: int) -> User | None: ...

    @abstractmethod
    def get_delivery_by_id(self, delivery_id: int) -> Delivery | None: ...

    @abstractmethod
    def get_active_delivery_by_courier(self, courier_id: int) -> Delivery | None:
        """Доставка курьера со статусом in_progress (если есть)."""

    @abstractmethod
    def save_delivery(self, order_id: int, courier_id: int) -> Delivery:
        """Создаёт доставку со статусом in_progress."""

    @abstractmethod
    def save_completion(self, delivery: Delivery, order: Order | None) -> None:
        """Сохраняет статус доставки и (если есть) заказа одной транзакцией."""


class SqlDeliveryRepository(DeliveryRepository):
    """Реализация на PostgreSQL через SQLAlchemy ORM."""

    def __init__(self, db: Session):
        self.db = db

    def get_order_by_id(self, order_id: int) -> Order | None:
        row = self.db.query(OrderModel).filter(OrderModel.id == order_id).first()
        return to_order(row)

    def get_user_by_id(self, user_id: int) -> User | None:
        row = self.db.query(UserModel).filter(UserModel.id == user_id).first()
        return to_user(row)

    def get_delivery_by_id(self, delivery_id: int) -> Delivery | None:
        row = self.db.query(DeliveryModel).filter(DeliveryModel.id == delivery_id).first()
        return to_delivery(row)

    def get_active_delivery_by_courier(self, courier_id: int) -> Delivery | None:
        row = (
            self.db.query(DeliveryModel)
            .filter(
                DeliveryModel.courier_id == courier_id,
                DeliveryModel.status == "in_progress",
            )
            .first()
        )
        return to_delivery(row)

    def save_delivery(self, order_id: int, courier_id: int) -> Delivery:
        row = DeliveryModel(order_id=order_id, courier_id=courier_id, status="in_progress")
        self.db.add(row)
        self.db.commit()
        self.db.refresh(row)
        return to_delivery(row)

    def save_completion(self, delivery: Delivery, order: Order | None) -> None:
        delivery_row = (
            self.db.query(DeliveryModel).filter(DeliveryModel.id == delivery.id).first()
        )
        delivery_row.status = delivery.status
        self.db.add(delivery_row)

        if order is not None:
            order_row = self.db.query(OrderModel).filter(OrderModel.id == order.id).first()
            order_row.status = order.status
            self.db.add(order_row)

        self.db.commit()


class InMemoryDeliveryRepository(DeliveryRepository):
    """Реализация в памяти (для тестов и запуска без СУБД)."""

    def __init__(self, store: MemoryStore | None = None):
        self.store = store or MemoryStore()

    def get_order_by_id(self, order_id: int) -> Order | None:
        return self.store.orders.get(order_id)

    def get_user_by_id(self, user_id: int) -> User | None:
        return self.store.users.get(user_id)

    def get_delivery_by_id(self, delivery_id: int) -> Delivery | None:
        return self.store.deliveries.get(delivery_id)

    def get_active_delivery_by_courier(self, courier_id: int) -> Delivery | None:
        for delivery in self.store.deliveries.values():
            if delivery.courier_id == courier_id and delivery.status == "in_progress":
                return delivery
        return None

    def save_delivery(self, order_id: int, courier_id: int) -> Delivery:
        delivery = Delivery(
            id=self.store.next_id(self.store.deliveries),
            order_id=order_id,
            courier_id=courier_id,
            status="in_progress",
        )
        self.store.deliveries[delivery.id] = delivery
        return delivery

    def save_completion(self, delivery: Delivery, order: Order | None) -> None:
        self.store.deliveries[delivery.id] = delivery
        if order is not None:
            self.store.orders[order.id] = order

"""
Слой бизнес-логики модуля delivery (Круги 1 и 2).
Не зависит от fastapi и sqlalchemy.
"""
from app.exceptions import ForbiddenError, NotFoundError
from app.modules.delivery.repository import DeliveryRepository


class DeliveryService:
    """Бизнес-правила доставки заказов."""

    def __init__(self, repository: DeliveryRepository):
        # Внедрение зависимости через конструктор (инверсия зависимостей).
        self.repository = repository

    def assign_courier(self, order_id: int, courier_id: int) -> int:
        """Назначает курьера на заказ и возвращает id доставки."""
        order = self.repository.get_order_by_id(order_id)
        if not order:
            raise NotFoundError("Заказ не найден")

        # Бизнес-правило: курьера можно назначить только на оплаченный заказ.
        if order.status != "paid":
            raise ValueError("Курьера можно назначить только на оплаченный заказ")

        courier = self.repository.get_user_by_id(courier_id)
        if not courier:
            raise NotFoundError("Курьер не найден")
        if courier.is_blocked:
            raise ForbiddenError("Курьер заблокирован и не может быть назначен")

        # Бизнес-правило: один курьер - одна активная доставка.
        if self.repository.get_active_delivery_by_courier(courier_id):
            raise ValueError("У курьера уже есть активная доставка")

        return self.repository.save_delivery(order_id, courier_id).id

    def complete_delivery(self, delivery_id: int) -> None:
        """Завершает доставку и помечает заказ как доставленный."""
        delivery = self.repository.get_delivery_by_id(delivery_id)
        if not delivery:
            raise NotFoundError("Доставка не найдена")
        if delivery.status == "completed":
            raise ValueError("Доставка уже завершена")
        if delivery.status == "cancelled":
            raise ValueError("Отменённую доставку нельзя завершить")

        delivery.status = "completed"

        order = self.repository.get_order_by_id(delivery.order_id)
        if order:
            order.status = "delivered"

        self.repository.save_completion(delivery, order)

"""
Инфраструктурный слой модуля menu (Круг 3).
Абстрактный интерфейс и две реализации: SQL и in-memory.
"""
from abc import ABC, abstractmethod

from sqlalchemy.orm import Session

from app.entities import MenuItem
from app.mappers import to_menu_item
from app.memory_store import MemoryStore
from app.models import MenuItem as MenuItemModel


class MenuRepository(ABC):
    """Интерфейс доступа к данным меню."""

    @abstractmethod
    def get_menu_item_by_id(self, item_id: int) -> MenuItem | None: ...

    @abstractmethod
    def get_menu_item_by_name(self, name: str) -> MenuItem | None: ...

    @abstractmethod
    def save_menu_item(self, name: str, price: float) -> MenuItem:
        """Создаёт доступное блюдо."""

    @abstractmethod
    def update_menu_item(self, item: MenuItem) -> None:
        """Сохраняет изменённые поля блюда."""


class SqlMenuRepository(MenuRepository):
    """Реализация на PostgreSQL через SQLAlchemy ORM."""

    def __init__(self, db: Session):
        self.db = db

    def get_menu_item_by_id(self, item_id: int) -> MenuItem | None:
        row = self.db.query(MenuItemModel).filter(MenuItemModel.id == item_id).first()
        return to_menu_item(row)

    def get_menu_item_by_name(self, name: str) -> MenuItem | None:
        row = self.db.query(MenuItemModel).filter(MenuItemModel.name == name).first()
        return to_menu_item(row)

    def save_menu_item(self, name: str, price: float) -> MenuItem:
        row = MenuItemModel(name=name, price=price, is_available=True)
        self.db.add(row)
        self.db.commit()
        self.db.refresh(row)
        return to_menu_item(row)

    def update_menu_item(self, item: MenuItem) -> None:
        row = self.db.query(MenuItemModel).filter(MenuItemModel.id == item.id).first()
        row.name = item.name
        row.price = item.price
        row.is_available = item.is_available
        self.db.add(row)
        self.db.commit()


class InMemoryMenuRepository(MenuRepository):
    """Реализация в памяти (для тестов и запуска без СУБД)."""

    def __init__(self, store: MemoryStore | None = None):
        self.store = store or MemoryStore()

    def get_menu_item_by_id(self, item_id: int) -> MenuItem | None:
        return self.store.menu_items.get(item_id)

    def get_menu_item_by_name(self, name: str) -> MenuItem | None:
        for item in self.store.menu_items.values():
            if item.name == name:
                return item
        return None

    def save_menu_item(self, name: str, price: float) -> MenuItem:
        item = MenuItem(
            id=self.store.next_id(self.store.menu_items),
            name=name,
            price=price,
            is_available=True,
        )
        self.store.menu_items[item.id] = item
        return item

    def update_menu_item(self, item: MenuItem) -> None:
        self.store.menu_items[item.id] = item

"""
Слой бизнес-логики модуля menu (Круги 1 и 2).
Не зависит от fastapi и sqlalchemy.
"""
from app.exceptions import NotFoundError
from app.modules.menu.repository import MenuRepository


class MenuService:
    """Бизнес-правила работы с меню."""

    def __init__(self, repository: MenuRepository):
        # Внедрение зависимости через конструктор (инверсия зависимостей).
        self.repository = repository

    def create_menu_item(self, name: str, price: float) -> int:
        """Создаёт блюдо и возвращает его id. ValueError, если название занято."""
        if self.repository.get_menu_item_by_name(name):
            raise ValueError("Блюдо с таким названием уже существует")
        return self.repository.save_menu_item(name, price).id

    def update_menu_item(
        self,
        item_id: int,
        name: str | None,
        price: float | None,
        is_available: bool | None,
    ) -> int:
        """Обновляет переданные поля блюда и возвращает его id."""
        item = self.repository.get_menu_item_by_id(item_id)
        if not item:
            raise NotFoundError("Блюдо не найдено")

        if name is not None:
            item.name = name
        if price is not None:
            item.price = price
        if is_available is not None:
            item.is_available = is_available

        self.repository.update_menu_item(item)
        return item.id

Инфраструктурный слой модуля orders (Круг 3).
Абстрактный интерфейс и две реализации: SQL и in-memory.
Класс отвечает только за ввод-вывод, бизнес-правил здесь нет.

from abc import ABC, abstractmethod

from sqlalchemy.orm import Session

from app.entities import MenuItem, Order, User
from app.mappers import to_menu_item, to_user
from app.memory_store import MemoryStore
from app.models import MenuItem as MenuItemModel
from app.models import Order as OrderModel
from app.models import User as UserModel


class OrderRepository(ABC):
    """Интерфейс доступа к данным, нужным сценарию оформления заказа."""

    @abstractmethod
    def get_user_by_id(self, user_id: int) -> User | None:
        """Извлечение пользователя по идентификатору."""

    @abstractmethod
    def get_menu_item_by_id(self, item_id: int) -> MenuItem | None:
        """Извлечение блюда из меню по идентификатору."""

    @abstractmethod
    def save_order_and_update_balance(
        self, user_id: int, item_id: int, quantity: int, new_balance: float
    ) -> int:
        """
        Создаёт заказ и сохраняет новый баланс пользователя одной транзакцией.
        Возвращает идентификатор созданного заказа.
        """


class SqlOrderRepository(OrderRepository):
    """Реализация на PostgreSQL через SQLAlchemy ORM."""

    def __init__(self, db: Session):
        self.db = db

    def get_user_by_id(self, user_id: int) -> User | None:
        row = self.db.query(UserModel).filter(UserModel.id == user_id).first()
        return to_user(row)

    def get_menu_item_by_id(self, item_id: int) -> MenuItem | None:
        row = self.db.query(MenuItemModel).filter(MenuItemModel.id == item_id).first()
        return to_menu_item(row)

    def save_order_and_update_balance(
        self, user_id: int, item_id: int, quantity: int, new_balance: float
    ) -> int:
        user_row = self.db.query(UserModel).filter(UserModel.id == user_id).first()
        user_row.balance = new_balance
        self.db.add(user_row)

        new_order = OrderModel(
            user_id=user_id,
            item_id=item_id,
            quantity=quantity,
            status="processing",
        )
        self.db.add(new_order)

        self.db.commit()  # UPDATE баланса и INSERT заказа - одна транзакция
        self.db.refresh(new_order)
        return new_order.id


class InMemoryOrderRepository(OrderRepository):
    """Реализация в памяти (для тестов и запуска без СУБД)."""

    def __init__(self, store: MemoryStore | None = None):
        self.store = store or MemoryStore()

    def get_user_by_id(self, user_id: int) -> User | None:
        return self.store.users.get(user_id)

    def get_menu_item_by_id(self, item_id: int) -> MenuItem | None:
        return self.store.menu_items.get(item_id)

    def save_order_and_update_balance(
        self, user_id: int, item_id: int, quantity: int, new_balance: float
    ) -> int:
        self.store.users[user_id].balance = new_balance
        order = Order(
            id=self.store.next_id(self.store.orders),
            user_id=user_id,
            item_id=item_id,
            quantity=quantity,
            status="processing",
        )
        self.store.orders[order.id] = order
        return order.id

        return self.repository.save_order_and_update_balance(
            user_id, item_id, quantity, new_balance

# Сириус.Еда — рефакторинг в Clean Architecture

Все четыре модуля (`users`, `menu`, `orders`, `delivery`) разделены на три слоя:
`router.py`, `service.py`, `repository.py`. Внешнее поведение API не изменилось.

## Что изменилось

| Было (MVP) | Стало |
| --- | --- |
| HTTP, бизнес-правила и SQL в одной функции `router.py` | Три файла на модуль: `router.py` -> `service.py` -> `repository.py` |
| Бизнес-логика работает с ORM-моделями | Бизнес-логика работает с сущностями `app/entities.py` (обычные dataclass) |
| `HTTPException` внутри бизнес-проверок | Сервис бросает `ValueError` / `NotFoundError` / `ForbiddenError`, роутер переводит их в 400 / 404 / 403 |
| Репозитория нет | Абстрактный интерфейс (ABC) в `repository.py` и две реализации: `Sql...` (PostgreSQL) и `InMemory...` |
| Сборка слоёв в каждом обработчике | Одно место сборки: `app/container.py` (composition root) |

Правило зависимостей: `service.py` не импортирует `fastapi` и `sqlalchemy`
(это проверяет тест `tests/test_architecture.py`).

## Структура

```text
app/
├── main.py            # точка входа
├── container.py       # сборка: репозиторий -> сервис (выбор реализации)
├── database.py        # подключение к PostgreSQL
├── models.py          # ORM-модели (SQLAlchemy)
├── entities.py        # сущности предметной области (dataclass)
├── mappers.py         # ORM-модель -> сущность
├── memory_store.py    # хранилище для in-memory репозиториев
├── exceptions.py      # доменные исключения (без fastapi)
├── http_errors.py     # исключение -> HTTP-статус
├── schemas.py         # Pydantic-схемы
└── modules/
    ├── users/     {router.py, service.py, repository.py}
    ├── menu/      {router.py, service.py, repository.py}
    ├── orders/    {router.py, service.py, repository.py}
    └── delivery/  {router.py, service.py, repository.py}
tests/                 # unit-тесты (без СУБД) и API-тест на in-memory
```

## Путь запроса через слои

```mermaid
sequenceDiagram
    participant C as Клиент
    participant R as router.py (API)
    participant S as service.py (бизнес-логика)
    participant P as repository.py (интерфейс)
    participant D as SqlRepository / InMemoryRepository
    C->>R: POST /orders/create
    R->>R: техническая валидация (quantity > 0)
    R->>S: create_order(user_id, item_id, quantity)
    S->>P: get_user_by_id, get_menu_item_by_id
    P->>D: реализация выбирается в container.py
    D-->>S: сущности User, MenuItem
    S->>S: бизнес-правила (доступность, баланс)
    S->>P: save_order_and_update_balance
    S-->>R: order_id или ValueError
    R-->>C: 200 {"status": "success", ...} или 400/403/404
```

```text

## Примечания

- Технические проверки (`quantity > 0`, `amount > 0`, формат телефона, цена и длина
  названия блюда) выполняются в роутере, бизнес-правила — в сервисе.
- Заказ создаётся со статусом `processing`, а курьера можно назначить только на
  заказ со статусом `paid` (так было в исходном MVP). Эндпоинта оплаты в проекте нет,
  поэтому для ручной проверки `delivery` статус заказа меняют в БД:
  `UPDATE orders SET status = 'paid' WHERE id = 1;`

HTTP -> router.py -> service.py -> repository.py (ABC) -> SQL / in-memory
        (Круг 4)     (Круги 1-2)    (Круг 3)
```

## Запуск

```bash
python3 -m venv .venv
source .venv/bin/activate
pip install -r requirements.txt
sudo docker-compose up -d db redis
uvicorn app.main:app --reload
```

Swagger: http://127.0.0.1:8000/docs (SQL-логи выводятся в терминал;
отключить: `SQL_ECHO=0`).

## Тесты

```bash
pytest -v
```

Тесты не требуют PostgreSQL: сервисы проверяются на `InMemory...` репозиториях.

## Замена реализации репозитория

Переключатель находится только в `app/container.py`, файлы `service.py` и
`router.py` не меняются:

```bash
REPO_BACKEND=memory uvicorn app.main:app --reload   # без PostgreSQL, данные в памяти
REPO_BACKEND=postgres uvicorn app.main:app --reload # по умолчанию
```

## Примечания

- Технические проверки (`quantity > 0`, `amount > 0`, формат телефона, цена и длина
  названия блюда) выполняются в роутере, бизнес-правила — в сервисе.
- Заказ создаётся со статусом `processing`, а курьера можно назначить только на
  заказ со статусом `paid` (так было в исходном MVP). Эндпоинта оплаты в проекте нет,
  поэтому для ручной проверки `delivery` статус заказа меняют в БД:
  `UPDATE orders SET status = 'paid' WHERE id = 1;`

"""
Инфраструктурный слой модуля users (Круг 3).
Абстрактный интерфейс и две реализации: SQL и in-memory.
Бизнес-правил здесь нет.
"""
from abc import ABC, abstractmethod

from sqlalchemy.orm import Session

from app.entities import User
from app.mappers import to_user
from app.memory_store import MemoryStore
from app.models import User as UserModel


class UserRepository(ABC):
    """Интерфейс доступа к данным пользователей."""

    @abstractmethod
    def get_user_by_id(self, user_id: int) -> User | None: ...

    @abstractmethod
    def get_user_by_email(self, email: str) -> User | None: ...

    @abstractmethod
    def save_user(self, email: str, phone: str | None) -> User:
        """Создаёт пользователя с нулевым балансом."""

    @abstractmethod
    def update_user_balance(self, user_id: int, new_balance: float) -> None: ...


class SqlUserRepository(UserRepository):
    """Реализация на PostgreSQL через SQLAlchemy ORM."""

    def __init__(self, db: Session):
        self.db = db

    def get_user_by_id(self, user_id: int) -> User | None:
        row = self.db.query(UserModel).filter(UserModel.id == user_id).first()
        return to_user(row)

    def get_user_by_email(self, email: str) -> User | None:
        row = self.db.query(UserModel).filter(UserModel.email == email).first()
        return to_user(row)

    def save_user(self, email: str, phone: str | None) -> User:
        row = UserModel(email=email, phone=phone, balance=0.0, is_blocked=False)
        self.db.add(row)
        self.db.commit()
        self.db.refresh(row)
        return to_user(row)

    def update_user_balance(self, user_id: int, new_balance: float) -> None:
        row = self.db.query(UserModel).filter(UserModel.id == user_id).first()
        row.balance = new_balance
        self.db.add(row)
        self.db.commit()


class InMemoryUserRepository(UserRepository):
    """Реализация в памяти (для тестов и запуска без СУБД)."""

    def __init__(self, store: MemoryStore | None = None):
        self.store = store or MemoryStore()

    def get_user_by_id(self, user_id: int) -> User | None:
        return self.store.users.get(user_id)

    def get_user_by_email(self, email: str) -> User | None:
        for user in self.store.users.values():
            if user.email == email:
                return user
        return None

    def save_user(self, email: str, phone: str | None) -> User:
        user = User(
            id=self.store.next_id(self.store.users),
            email=email,
            phone=phone,
            balance=0.0,
            is_blocked=False,
        )
        self.store.users[user.id] = user
        return user

    def update_user_balance(self, user_id: int, new_balance: float) -> None:
        self.store.users[user_id].balance = new_balance
'''

"""
Слой бизнес-логики модуля users (Круги 1 и 2).
Не зависит от fastapi и sqlalchemy.
"""
from app.entities import User
from app.exceptions import ForbiddenError, NotFoundError
from app.modules.users.repository import UserRepository

MAX_TOPUP_AMOUNT = 50000


class UserService:
    """Бизнес-правила работы с пользователями."""

    def __init__(self, repository: UserRepository):
        # Внедрение зависимости через конструктор (инверсия зависимостей).
        self.repository = repository

    def register_user(self, email: str, phone: str | None) -> User:
        """Регистрирует пользователя. ValueError, если email уже занят."""
        if self.repository.get_user_by_email(email):
            raise ValueError("Пользователь с таким email уже зарегистрирован")
        return self.repository.save_user(email, phone)

    def top_up_balance(self, user_id: int, amount: float) -> float:
        """Пополняет баланс и возвращает новый баланс."""
        user = self.repository.get_user_by_id(user_id)
        if not user:
            raise NotFoundError("Пользователь не найден")
        if user.is_blocked:
            raise ForbiddenError("Пользователь заблокирован. Пополнение невозможно")
        if amount > MAX_TOPUP_AMOUNT:
            raise ValueError("Максимальная сумма пополнения — 50000 рублей")

        new_balance = user.balance + amount
        self.repository.update_user_balance(user_id, new_balance)
        return new_balance
'''

"""
Инфраструктурный слой модуля orders (Круг 3).
Абстрактный интерфейс и две реализации: SQL и in-memory.
Класс отвечает только за ввод-вывод, бизнес-правил здесь нет.
"""
from abc import ABC, abstractmethod

from sqlalchemy.orm import Session

from app.entities import MenuItem, Order, User
from app.mappers import to_menu_item, to_user
from app.memory_store import MemoryStore
from app.models import MenuItem as MenuItemModel
from app.models import Order as OrderModel
from app.models import User as UserModel


class OrderRepository(ABC):
    """Интерфейс доступа к данным, нужным сценарию оформления заказа."""

    @abstractmethod
    def get_user_by_id(self, user_id: int) -> User | None:
        """Извлечение пользователя по идентификатору."""

    @abstractmethod
    def get_menu_item_by_id(self, item_id: int) -> MenuItem | None:
        """Извлечение блюда из меню по идентификатору."""

    @abstractmethod
    def save_order_and_update_balance(
        self, user_id: int, item_id: int, quantity: int, new_balance: float
    ) -> int:
        """
        Создаёт заказ и сохраняет новый баланс пользователя одной транзакцией.
        Возвращает идентификатор созданного заказа.
        """


class SqlOrderRepository(OrderRepository):
    """Реализация на PostgreSQL через SQLAlchemy ORM."""

    def __init__(self, db: Session):
        self.db = db

    def get_user_by_id(self, user_id: int) -> User | None:
        row = self.db.query(UserModel).filter(UserModel.id == user_id).first()
        return to_user(row)

    def get_menu_item_by_id(self, item_id: int) -> MenuItem | None:
        row = self.db.query(MenuItemModel).filter(MenuItemModel.id == item_id).first()
        return to_menu_item(row)

    def save_order_and_update_balance(
        self, user_id: int, item_id: int, quantity: int, new_balance: float
    ) -> int:
        user_row = self.db.query(UserModel).filter(UserModel.id == user_id).first()
        user_row.balance = new_balance
        self.db.add(user_row)

        new_order = OrderModel(
            user_id=user_id,
            item_id=item_id,
            quantity=quantity,
            status="processing",
        )
        self.db.add(new_order)

        self.db.commit()  # UPDATE баланса и INSERT заказа - одна транзакция
        self.db.refresh(new_order)
        return new_order.id


class InMemoryOrderRepository(OrderRepository):
    """Реализация в памяти (для тестов и запуска без СУБД)."""

    def __init__(self, store: MemoryStore | None = None):
        self.store = store or MemoryStore()

    def get_user_by_id(self, user_id: int) -> User | None:
        return self.store.users.get(user_id)

    def get_menu_item_by_id(self, item_id: int) -> MenuItem | None:
        return self.store.menu_items.get(item_id)

    def save_order_and_update_balance(
        self, user_id: int, item_id: int, quantity: int, new_balance: float
    ) -> int:
        self.store.users[user_id].balance = new_balance
        order = Order(
            id=self.store.next_id(self.store.orders),
            user_id=user_id,
            item_id=item_id,
            quantity=quantity,
            status="processing",
        )
        self.store.orders[order.id] = order
        return order.id

"""
Слой бизнес-логики модуля orders (Круги 1 и 2).

from app.exceptions import NotFoundError
from app.modules.orders.repository import OrderRepository


class OrderService:
    """Бизнес-правила обработки заказов."""

    def __init__(self, repository: OrderRepository):
        # Внедрение зависимости через конструктор (инверсия зависимостей).
        self.repository = repository

    def create_order(self, user_id: int, item_id: int, quantity: int) -> int:
        """
        Сценарий оформления заказа. Возвращает идентификатор заказа.
        Возбуждает ValueError (или его наследников) при нарушении бизнес-правил.

        if not user:

        if user.balance < total_cost:
            raise ValueError("Недостаточно средств на балансе студента")

        new_balance = user.balance - total_cost
        return self.repository.save_order_and_update_balance(
            user_id, item_id, quantity, new_balance
        )

"""
Инфраструктурный слой модуля delivery (Круг 3).
Абстрактный интерфейс и две реализации: SQL и in-memory.
"""
from abc import ABC, abstractmethod

from sqlalchemy.orm import Session

from app.entities import Delivery, Order, User
from app.mappers import to_delivery, to_order, to_user
from app.memory_store import MemoryStore
from app.models import Delivery as DeliveryModel
from app.models import Order as OrderModel
from app.models import User as UserModel


class DeliveryRepository(ABC):
    """Интерфейс доступа к данным доставки."""

    @abstractmethod
    def get_order_by_id(self, order_id: int) -> Order | None: ...

    @abstractmethod
    def get_user_by_id(self, user_id: int) -> User | None: ...

    @abstractmethod
    def get_delivery_by_id(self, delivery_id: int) -> Delivery | None: ...

    @abstractmethod
    def get_active_delivery_by_courier(self, courier_id: int) -> Delivery | None:
        """Доставка курьера со статусом in_progress (если есть)."""

    @abstractmethod
    def save_delivery(self, order_id: int, courier_id: int) -> Delivery:
        """Создаёт доставку со статусом in_progress."""

    @abstractmethod
    def save_completion(self, delivery: Delivery, order: Order | None) -> None:
        """Сохраняет статус доставки и (если есть) заказа одной транзакцией."""


class SqlDeliveryRepository(DeliveryRepository):
    """Реализация на PostgreSQL через SQLAlchemy ORM."""

    def __init__(self, db: Session):
        self.db = db

    def get_order_by_id(self, order_id: int) -> Order | None:
        row = self.db.query(OrderModel).filter(OrderModel.id == order_id).first()
        return to_order(row)

    def get_user_by_id(self, user_id: int) -> User | None:
        row = self.db.query(UserModel).filter(UserModel.id == user_id).first()
        return to_user(row)

    def get_delivery_by_id(self, delivery_id: int) -> Delivery | None:
        row = self.db.query(DeliveryModel).filter(DeliveryModel.id == delivery_id).first()
        return to_delivery(row)

    def get_active_delivery_by_courier(self, courier_id: int) -> Delivery | None:
        row = (
            self.db.query(DeliveryModel)
            .filter(
                DeliveryModel.courier_id == courier_id,
                DeliveryModel.status == "in_progress",
            )
            .first()
        )
        return to_delivery(row)

    def save_delivery(self, order_id: int, courier_id: int) -> Delivery:
        row = DeliveryModel(order_id=order_id, courier_id=courier_id, status="in_progress")
        self.db.add(row)
        self.db.commit()
        self.db.refresh(row)
        return to_delivery(row)

    def save_completion(self, delivery: Delivery, order: Order | None) -> None:
        delivery_row = (
            self.db.query(DeliveryModel).filter(DeliveryModel.id == delivery.id).first()
        )
        delivery_row.status = delivery.status
        self.db.add(delivery_row)

        if order is not None:
            order_row = self.db.query(OrderModel).filter(OrderModel.id == order.id).first()
            order_row.status = order.status
            self.db.add(order_row)

        self.db.commit()


class InMemoryDeliveryRepository(DeliveryRepository):
    """Реализация в памяти (для тестов и запуска без СУБД)."""

    def __init__(self, store: MemoryStore | None = None):
        self.store = store or MemoryStore()

    def get_order_by_id(self, order_id: int) -> Order | None:
        return self.store.orders.get(order_id)

    def get_user_by_id(self, user_id: int) -> User | None:
        return self.store.users.get(user_id)

    def get_delivery_by_id(self, delivery_id: int) -> Delivery | None:
        return self.store.deliveries.get(delivery_id)

    def get_active_delivery_by_courier(self, courier_id: int) -> Delivery | None:
        for delivery in self.store.deliveries.values():
            if delivery.courier_id == courier_id and delivery.status == "in_progress":
                return delivery
        return None

    def save_delivery(self, order_id: int, courier_id: int) -> Delivery:
        delivery = Delivery(
            id=self.store.next_id(self.store.deliveries),
            order_id=order_id,
            courier_id=courier_id,
            status="in_progress",
        )
        self.store.deliveries[delivery.id] = delivery
        return delivery

    def save_completion(self, delivery: Delivery, order: Order | None) -> None:
        self.store.deliveries[delivery.id] = delivery
        if order is not None:
            self.store.orders[order.id] = order
'''

"""
Слой бизнес-логики модуля delivery (Круги 1 и 2).
Не зависит от fastapi и sqlalchemy.
"""
from app.exceptions import ForbiddenError, NotFoundError
from app.modules.delivery.repository import DeliveryRepository


class DeliveryService:
    """Бизнес-правила доставки заказов."""

    def __init__(self, repository: DeliveryRepository):
        # Внедрение зависимости через конструктор (инверсия зависимостей).
        self.repository = repository

    def assign_courier(self, order_id: int, courier_id: int) -> int:
        """Назначает курьера на заказ и возвращает id доставки."""
        order = self.repository.get_order_by_id(order_id)
        if not order:
            raise NotFoundError("Заказ не найден")

        # Бизнес-правило: курьера можно назначить только на оплаченный заказ.
        if order.status != "paid":
            raise ValueError("Курьера можно назначить только на оплаченный заказ")

        courier = self.repository.get_user_by_id(courier_id)
        if not courier:
            raise NotFoundError("Курьер не найден")
        if courier.is_blocked:
            raise ForbiddenError("Курьер заблокирован и не может быть назначен")

        # Бизнес-правило: один курьер - одна активная доставка.
        if self.repository.get_active_delivery_by_courier(courier_id):
            raise ValueError("У курьера уже есть активная доставка")

        return self.repository.save_delivery(order_id, courier_id).id

    def complete_delivery(self, delivery_id: int) -> None:
        """Завершает доставку и помечает заказ как доставленный."""
        delivery = self.repository.get_delivery_by_id(delivery_id)
        if not delivery:
            raise NotFoundError("Доставка не найдена")
        if delivery.status == "completed":
            raise ValueError("Доставка уже завершена")
        if delivery.status == "cancelled":
            raise ValueError("Отменённую доставку нельзя завершить")

        delivery.status = "completed"

        order = self.repository.get_order_by_id(delivery.order_id)
        if order:
            order.status = "delivered"

        self.repository.save_completion(delivery, order)
'''

import ast
from pathlib import Path

FORBIDDEN = ("fastapi", "sqlalchemy")
MODULES_DIR = Path(__file__).resolve().parent.parent / "app" / "modules"


def test_services_do_not_import_frameworks():
    services = sorted(MODULES_DIR.glob("*/service.py"))
    assert len(services) == 4

    for path in services:
        tree = ast.parse(path.read_text(encoding="utf-8"))
        for node in ast.walk(tree):
            if isinstance(node, ast.Import):
                names = [alias.name for alias in node.names]
            elif isinstance(node, ast.ImportFrom):
                names = [node.module or ""]
            else:
                continue
            for name in names:
                assert not name.startswith(FORBIDDEN), f"{path.name}: запрещённый импорт {name}"
